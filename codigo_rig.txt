// Medidas base del placeholder (en metros)
const M = {
  caderaSentado: 0.5,
  caderaParado: 0.95,
  radioTorso: 0.16,
  largoTorso: 0.3,
  altoHombro: 0.55,
  anchoHombro: 0.21,
  altoCuello: 0.64,
  radioCabeza: 0.12,
  largoBrazo: 0.27,
  largoAntebrazo: 0.25,
  radioBrazo: 0.045,
};

const loader = new GLTFLoader();

// Modelos precargados (ruta -> gltf). Se llenan con precargarModelos() antes de armar el elenco,
// asÃ­ los personajes se construyen con su esqueleto de una vez y los sistemas que cuelgan cosas
// de las manos (cartelito, papelito, lapicera) ya encuentran las articulaciones definitivas.
const cacheModelos = new Map();

const _v1 = new THREE.Vector3();
const _v2 = new THREE.Vector3();
const _v3 = new THREE.Vector3();
const _v4 = new THREE.Vector3();
const _v5 = new THREE.Vector3();

export async function precargarModelos(rutas, alProgresar) {
  let listos = 0;
  await Promise.all(
    [...new Set(rutas)].map(
      (ruta) =>
        new Promise((resolver) => {
          loader.load(
            ruta,
            (gltf) => {
              cacheModelos.set(ruta, gltf);
              alProgresar?.(++listos, rutas.length);
              resolver();
            },
            undefined,
            (err) => {
              console.warn(`No se pudo cargar ${ruta}; ese personaje queda como placeholder.`, err);
              alProgresar?.(++listos, rutas.length);
              resolver();
            },
          );
        }),
    ),
  );
}

/**
 * FÃ¡brica de personajes.
 *   crearPersonaje({ tipo: 'placeholder', ...config })
 *   crearPersonaje({ tipo: 'glb', ruta: '/modelos/x.glb', ... })
 * Siempre devuelve un Personaje con la misma interfaz (raiz, animador, mostrarDedos, etc.).
 */
export function crearPersonaje(config, nombre = 'personaje') {
  const p = new Personaje(config, nombre);
  const gltf = config.modelo && cacheModelos.get(config.modelo.ruta);
  if (gltf && config.modelo.rig !== false) {
    // Modelo estÃ¡tico + esqueleto generado por cÃ³digo (ver AutoRig.js)
    try {
      p.construirConModelo(config, gltf);
      return p;
    } catch (e) {
      console.warn(`No se pudo armar el esqueleto de ${config.modelo.ruta}; queda el placeholder.`, e);
      p.raiz.clear();
      p.rig = {};
    }
  }
  if (config.tipo === 'glb') p.cargarGLB(config);
  else p.construirPlaceholder(config);
  // Modelo estÃ¡tico sin esqueleto como "piel" (modo viejo: rig: false)
  if (config.modelo && config.modelo.rig === false) p.cargarModeloEstatico(config.modelo);
  return p;
}

  // ---------- Modelo con esqueleto automÃ¡tico ----------
  construirConModelo(cfg, gltf) {
    const m = cfg.modelo;
    const sentado = cfg.postura === 'sentado';
    const { rig, malla, contorno: bordes, articulaciones, manos } = autoRig(this.raiz, gltf.scene, m);
    this.manos = manos; // dedos articulados (null si no se detectaron: queda la mano postiza)
    this.rig = rig;
    this.malla = malla;
    this.mallaContorno = bordes;
    this.articulaciones = articulaciones;
    this.conModelo = true;

    // Lapicera (y mano postiza de respaldo si no se detectaron los dedos del modelo)
    const matPiel = toon(m.piel ?? cfg.piel ?? 0xeec29c);
    for (const [lado, signo] of [['brazoI', -1], ['brazoD', 1]]) {
      const br = rig[lado];
      if (!manos) this.construirManoPostiza(br, signo, matPiel);
      if (cfg.lapicera || cfg.soloBrazos || m.lapicera) {
        const lapicera = new THREE.Mesh(new THREE.CylinderGeometry(0.005, 0.005, 0.12, 6), toon(0x2050c0));
        lapicera.position.set(0.01 * signo, -0.075, -0.03);
        lapicera.rotation.x = Math.PI / 2;
        lapicera.visible = false;
        br.muneca.add(lapicera);
        br.lapicera = lapicera;
      }
    }

    // VisiÃ³n, highlight y oclusiÃ³n usan una caja simple en vez de la malla con piel (mucho mÃ¡s barato)
    const caja = sentado
      ? new THREE.Box3(new THREE.Vector3(-0.28, 0.35, -0.45), new THREE.Vector3(0.28, 1.32, 0.25))
      : new THREE.Box3(new THREE.Vector3(-0.3, 0, -0.22), new THREE.Vector3(0.3, m.altura, 0.22));
    const inversa = new THREE.Matrix4();
    const rayoLocal = new THREE.Ray();
    const punto = new THREE.Vector3();
    malla.raycast = (raycaster, intersecciones) => {
      inversa.copy(this.raiz.matrixWorld).invert();
      rayoLocal.copy(raycaster.ray).applyMatrix4(inversa);
      if (!rayoLocal.intersectBox(caja, punto)) return;
      punto.applyMatrix4(this.raiz.matrixWorld);
      const distancia = raycaster.ray.origin.distanceTo(punto);
      if (distancia < raycaster.near || distancia > raycaster.far) return;
      intersecciones.push({ distance: distancia, point: punto.clone(), object: malla });
    };

    // Postura: sentado = cadera a la altura del asiento, muslos adelante y piernas abajo
    if (sentado) {
      rig.cadera.position.y = m.alturaCadera ?? 0.52;
      rig.cadera.position.z += m.offsetCaderaZ ?? 0.04;
      // Torso mÃ¡s alto (sentarse derecho): sube el tronco, la cabeza y los brazos; las piernas quedan
      if (m.elevarTorso) rig.torso.position.y += m.elevarTorso;
      for (const [lado, signo] of [['I', -1], ['D', 1]]) {
        rig['pierna' + lado].rotation.set(Math.PI / 2 - 0.05, 0, -signo * 0.06);
        rig['rodilla' + lado].rotation.set(-Math.PI / 2 + 0.12, 0, signo * 0.04);
        rig['tobillo' + lado].rotation.set(-0.05, 0, 0);
      }
    }
    if (cfg.escala) this.raiz.scale.setScalar(cfg.escala);
    // Primera persona: la cabeza no se dibuja (la cÃ¡mara estÃ¡ ahÃ­)
    if (cfg.soloBrazos) rig.cabeza.scale.setScalar(0.001);
    this.animador.poseInicial(sentado ? (cfg.soloBrazos ? 'manosEnBanco' : 'sentado') : 'parado');
  }

  // ---------- Mano de seÃ±as (Sofi): agrandada y con la palma hacia el jugador ----------
  estadoManoSenia(lado) {
    this._manoSenia ??= {
      I: { escala: 1, objetivo: 1, giro: 0, punto: null },
      D: { escala: 1, objetivo: 1, giro: 0, punto: null },
    };
    return this._manoSenia[lado === 'brazoI' || lado === 'I' ? 'I' : 'D'];
  }

  /** Agranda (o devuelve a su tamaÃ±o) la mano del modelo, con transiciÃ³n suave. */
  escalarMano(lado, factor) {
    this.estadoManoSenia(lado).objetivo = factor;
  }

  /** Gira la muÃ±eca para que la palma mire a un punto del mundo (null = sin girar). */
  orientarPalma(lado, punto) {
    const e = this.estadoManoSenia(lado);
    e.punto = punto ? (e.punto ?? new THREE.Vector3()).copy(punto) : null;
  }

  actualizarManoSenia(dt) {
    if (!this._manoSenia) return;
    const k = 1 - Math.exp(-8 * dt);
    for (const lado of ['I', 'D']) {
      const e = this._manoSenia[lado];
      const br = this.rig['brazo' + lado];
      if (!br?.manoModelo) continue;
      e.escala += (e.objetivo - e.escala) * k;
      br.manoModelo.scale.setScalar(e.escala);
      let giro = 0;
      if (e.punto) {
        // La palma mira hacia -x local en la mano derecha y +x en la izquierda (pose de atadura).
        // Se gira alrededor del eje del antebrazo (y local) hasta que apunte al punto.
        br.muneca.updateWorldMatrix(true, false);
        const m = br.muneca.matrixWorld;
        const eje = _v1.setFromMatrixColumn(m, 1).normalize();
        const centro = _v2.setFromMatrixPosition(m);
        const normal = _v3.set(lado === 'D' ? -1 : 1, 0, 0).transformDirection(m);
        const meta = _v4.copy(e.punto).sub(centro);
        normal.addScaledVector(eje, -normal.dot(eje)).normalize();
        meta.addScaledVector(eje, -meta.dot(eje));
        if (meta.lengthSq() > 1e-6) {
          meta.normalize();
          giro = Math.atan2(eje.dot(_v5.crossVectors(normal, meta)), normal.dot(meta));
        }
      }
      e.giro += (giro - e.giro) * k;
      br.muneca.rotation.y += e.giro;
    }
  }

  // Mano "postiza" con dedos exagerados: sÃ³lo si no se pudieron detectar los dedos del modelo
  construirManoPostiza(br, signo, matPiel) {
    const mano = new THREE.Mesh(new THREE.SphereGeometry(0.05, 12, 10), matPiel);
    mano.scale.set(0.9, 1.15, 0.65);
    mano.position.y = -0.045;
    mano.visible = false;
    contorno(mano, 0.006);
    br.muneca.add(mano);
    br.mano = mano;
    for (let i = 0; i < 4; i++) {
      const dedo = new THREE.Mesh(new THREE.CapsuleGeometry(0.012, 0.065, 3, 6), matPiel);
      dedo.position.set((i - 1.5) * 0.022 * -signo, -0.115, 0);
      dedo.visible = false;
      contorno(dedo, 0.005);
      br.muneca.add(dedo);
      br.dedos.push(dedo);
    }
  }

  // ---------- Modelo GLB ----------
  cargarGLB(cfg) {
    loader.load(
      cfg.ruta,
      (gltf) => {
        const modelo = gltf.scene;
        modelo.traverse((o) => {
          if (o.isMesh) o.castShadow = true;
        });
        if (cfg.escala) modelo.scale.setScalar(cfg.escala);
        this.raiz.add(modelo);
        // Mapear huesos del modelo a las articulaciones que usa el animador
        if (cfg.huesos) {
          for (const [articulacion, nombreHueso] of Object.entries(cfg.huesos)) {
            const hueso = modelo.getObjectByName(nombreHueso);
            if (!hueso) continue;
            if (articulacion.startsWith('hombro') || articulacion.startsWith('codo')) {
              const lado = articulacion.endsWith('I') ? 'brazoI' : 'brazoD';
              this.rig[lado] ??= { dedos: [] };
              this.rig[lado][articulacion.startsWith('hombro') ? 'hombro' : 'codo'] = hueso;
            } else {
              this.rig[articulacion] = hueso;
            }
          }
        }
        if (gltf.animations.length) {
          this.mixer = new THREE.AnimationMixer(modelo);
          const clip = THREE.AnimationClip.findByName(gltf.animations, cfg.animacion) ?? gltf.animations[0];
          this.mixer.clipAction(clip).play();
        }
        this.animador.capturarReposo();
      },
      undefined,
      (err) => console.warn(`No se pudo cargar ${cfg.ruta}; queda vacÃ­o.`, err),
    );
  }

 * los generadores de personajes 3D).

 * 1. Normaliza la malla: altura en metros, mirando a -Z, pies en y = 0.
 * 2. Detecta las articulaciones cortando el cuerpo en rebanadas horizontales
 *    (dÃ³nde se separan los brazos del torso, dÃ³nde estÃ¡ el cuello, las piernas, etc.).
 * 3. Arma un esqueleto con LOS MISMOS NOMBRES Y CONVENCIONES que el personaje placeholder
 *    (cadera, torso, cuello, cabeza, brazoI/D.{hombro,codo,muneca}, piernaI/D...), asÃ­ el
 *    Animador y todos los sistemas funcionan igual con el modelo.
 * 4. Calcula los pesos de piel por distancia a cada hueso (con reglas por zona).
 *
 * Los brazos se "atan" en la orientaciÃ³n de la pose A, de modo que con rotaciÃ³n 0 el brazo

function geometriaNormalizada(escena, { altura, rotacionY = 0 }) {
  escena.updateMatrixWorld(true);
  const mallas = [];
  escena.traverse((o) => o.isMesh && mallas.push(o));
  if (!mallas.length) throw new Error('El modelo no tiene mallas');
  // Se usa la malla mÃ¡s grande (los generadores exportan una sola)
  mallas.sort((m1, m2) => m2.geometry.attributes.position.count - m1.geometry.attributes.position.count);
  const malla = mallas[0];
  const g = malla.geometry.clone();
  g.applyMatrix4(malla.matrixWorld);
  if (rotacionY) g.rotateY(rotacionY);
  g.computeBoundingBox();
  const bb = g.boundingBox;
  const s = altura / (bb.max.y - bb.min.y);
  g.translate(0, -bb.min.y, 0);
  g.scale(s, s, s);
  // Centro horizontal: x del eje de simetrÃ­a, z del torso (para que se siente donde va el placeholder)
  const pos = g.attributes.position;
  let sx = 0;
  let sz = 0;
  let n = 0;
  for (let i = 0; i < pos.count; i++) {
    const y = pos.getY(i);
    if (y > altura * 0.5 && y < altura * 0.75 && Math.abs(pos.getX(i)) < altura * 0.12) {
      sx += pos.getX(i);
      sz += pos.getZ(i);
      n++;
    }
  }
  g.translate(-(n ? sx / n : 0), 0, -(n ? sz / n : 0));
  if (!g.attributes.normal) g.computeVertexNormals();
  return { geometria: g, material: malla.material };
}

// Agrupa los vÃ©rtices de una franja de altura en "islas" separadas en x (brazos, torso, piernas)
function rebanada(pos, y, banda, hueco) {
  const pts = [];
  for (let i = 0; i < pos.count; i++) {
    const py = pos.getY(i);
    if (Math.abs(py - y) < banda) pts.push(i);
  }
  pts.sort((i, j) => pos.getX(i) - pos.getX(j));
  const islas = [];
  let actual = null;
  let ultimoX = -Infinity;
  for (const i of pts) {
    const x = pos.getX(i);
    if (!actual || x - ultimoX > hueco) {
      actual = { min: x, max: x, sz: 0, n: 0 };
      islas.push(actual);
    }
    actual.max = x;
    actual.sz += pos.getZ(i);
    actual.n++;
    ultimoX = x;
  }
  for (const isla of islas) {
    isla.centro = (isla.min + isla.max) / 2;
    isla.z = isla.sz / isla.n;
    isla.ancho = isla.max - isla.min;
  }
  // Se descartan islas mÃ­nimas (ruido de la malla)
  return islas.filter((isla) => isla.n >= 3);
}

function ajusteLineal(puntos) {
  // x = a + b*y por mÃ­nimos cuadrados
  let sy = 0, sx = 0, syy = 0, sxy = 0;
  const n = puntos.length;
  for (const p of puntos) {
    sy += p.y; sx += p.x; syy += p.y * p.y; sxy += p.x * p.y;
  }
  const b = (n * sxy - sx * sy) / Math.max(1e-9, n * syy - sy * sy);
  return { a: (sx - b * sy) / n, b };
}

// ---------- Esqueleto ----------
function hueso(nombre, padre, posMundo, posPadreMundo) {
  const b = new THREE.Bone();
  b.name = nombre;
  if (padre) {
    padre.add(b);
    padre.updateMatrixWorld(true);
    b.position.copy(padre.worldToLocal(posMundo.clone()));
  } else b.position.copy(posMundo).sub(posPadreMundo ?? new THREE.Vector3());
  b.updateMatrixWorld(true);
  return b;
}

// Orienta un hueso para que su -Y local apunte hacia "objetivo" (en mundo)
function apuntar(b, objetivo) {
  const desde = b.getWorldPosition(_a);
  const dir = _b.subVectors(objetivo, desde).normalize();
  const qMundo = new THREE.Quaternion().setFromUnitVectors(ABAJO, dir);
  const qPadre = b.parent.getWorldQuaternion(new THREE.Quaternion());
  b.quaternion.copy(qPadre.invert().multiply(qMundo));
  b.updateMatrixWorld(true);
}

function crearEsqueleto(raiz, art, dedos) {

function distSegmento(p, a, b) {
  const ab = _b.subVectors(b, a);
  const t = THREE.MathUtils.clamp(_a.subVectors(p, a).dot(ab) / Math.max(1e-9, ab.lengthSq()), 0, 1);
  return p.distanceTo(_a.copy(a).addScaledVector(ab, t));
}

      if (zona === 'pierna' && (p.y > art.cintura.y || p.x * signo < -H * 0.02)) continue;
      if (zona === 'cabeza' && p.y < art.cuello.y) continue;
      if (zona === 'cuello' && (p.y < art.cuello.y - H * 0.05 || p.y > art.cabeza.y + H * 0.06)) continue;
      const d = distSegmento(p, a, z) / r;
      cand.push([idx(b), 1 / (Math.pow(d, 5) + 1e-4), zona]);
    }
    cand.sort((c1, c2) => c2[1] - c1[1]);
    // Pollera / entrepierna: mezcla con la cadera para que la tela no se rompa al caminar
    // (sÃ³lo si lo que domina es una pierna: antes se mezclaba tambiÃ©n en las manos, que en pose A
    // quedan a la altura de la pelvis, y al mover los brazos se estiraban hacia la cintura)
    if (p.y > art.separacionPiernas - H * 0.02 && p.y < art.cintura.y && cand[0]?.[2] === 'pierna' && Math.abs(p.x) < H * 0.2) {
      const mezcla = art.conPollera ? 0.55 : 0.3;
      const total = cand.slice(0, 4).reduce((s, c) => s + c[1], 0);
      for (const c of cand) c[1] = (c[1] / total) * (1 - mezcla);
      const cad = cand.find((c) => c[0] === idx(rig.cadera));
      if (cad) cad[1] += mezcla;
      else cand.push([idx(rig.cadera), mezcla, 'tronco']);
      cand.sort((c1, c2) => c2[1] - c1[1]);
    }
    let total = 0;
    for (let k = 0; k < 4 && k < cand.length; k++) total += cand[k][1];
    for (let k = 0; k < 4; k++) {
      if (k < cand.length) {
        indices[i * 4 + k] = cand[k][0];
        pesos[i * 4 + k] = cand[k][1] / total;
      }
    }
  }
  g.setAttribute('skinIndex', new THREE.Uint16BufferAttribute(indices, 4));
  g.setAttribute('skinWeight', new THREE.Float32BufferAttribute(pesos, 4));

  // Dedos: reparten el peso de la mano entre las tres falanges de cada dedo
  for (const lado of ['I', 'D']) {
    const det = dedos?.[lado];
    const br = rig['brazo' + lado];
    if (det && br.dedosHuesos) pesosDedos(g, det, br.dedosHuesos, idx, br.manoModelo, det.largo);
  }
}

// Normales promediadas por posiciÃ³n: el contorno no se abre en las costuras de la textura
function normalesSuaves(g) {
  const pos = g.attributes.position;
  const nor = g.attributes.normal;
  const mapa = new Map();
  const clave = (i) => `${Math.round(pos.getX(i) * 2000)},${Math.round(pos.getY(i) * 2000)},${Math.round(pos.getZ(i) * 2000)}`;
  for (let i = 0; i < pos.count; i++) {
    const k = clave(i);
    const acc = mapa.get(k) ?? [0, 0, 0];
    acc[0] += nor.getX(i); acc[1] += nor.getY(i); acc[2] += nor.getZ(i);
    mapa.set(k, acc);
  }
  const datos = new Float32Array(pos.count * 3);
  for (let i = 0; i < pos.count; i++) {
    const acc = mapa.get(clave(i));
    _v.set(acc[0], acc[1], acc[2]).normalize();
    datos.set([_v.x, _v.y, _v.z], i * 3);
  }
  return new THREE.BufferAttribute(datos, 3);
}

export function materialContornoPiel(grosor, color = COLOR_CONTORNO) {

}

function recortarTorso(g, art, H) {
  const pos = g.attributes.position;

function preparar(escena, cfg) {
  const clave = `${cfg.ruta}|${cfg.altura}|${cfg.rotacionY ?? 0}|${cfg.soloTorso ? 'torso' : 'completo'}`;
  const guardado = cachePreparados.get(clave);
  if (guardado) return guardado;

  const H = cfg.altura;
  const { geometria: g, material: original } = geometriaNormalizada(escena, cfg);
  const art = detectarArticulaciones(g, H);

  const dedos = {};
  for (const lado of ['I', 'D']) {
    const diag = [];
    dedos[lado] = detectarDedos(g.attributes.position, art, lado, H, diag);
    if (!dedos[lado]) console.warn(`${cfg.ruta}: no se pudieron detectar los dedos (${lado}): ${diag.join('; ')}. La mano queda rÃ­gida.`);
  }

  // Esqueleto temporal: fija el orden de los huesos y permite calcular los pesos una Ãºnica vez
  const temporal = new THREE.Group();
  const rigTmp = crearEsqueleto(temporal, art, dedos);
  const huesosTmp = [];
  rigTmp.cadera.traverse((o) => o.isBone && huesosTmp.push(o));
  calcularPesos(g, art, rigTmp, huesosTmp, H, dedos);

  // Regiones de pelo y remera (para recolorear); si falla, el modelo queda con sus colores
  let regiones = null;
  try {
    regiones = clasificarRegiones(g, art, H, crearMuestreador(original.map));
    if (regiones) g.setAttribute('region', regiones.region);
  } catch (e) {
    console.warn(`${cfg.ruta}: no se pudieron clasificar pelo y remera.`, e);
  }

  if (cfg.soloTorso) recortarTorso(g, art, H);

  // Contorno: misma piel y esqueleto, con normales suaves
  const gc = new THREE.BufferGeometry();
  gc.setIndex(g.index);
  for (const nombre of ['position', 'skinIndex', 'skinWeight']) gc.setAttribute(nombre, g.attributes[nombre]);
  gc.setAttribute('normal', normalesSuaves(g));

  const listo = { g, gc, art, dedos, original, H, regiones, nHuesos: huesosTmp.length };
  cachePreparados.set(clave, listo);
  return listo;
}

/**

  raiz.rotation.set(0, 0, 0);
  raiz.scale.set(1, 1, 1);
  raiz.updateMatrixWorld(true);

  const huesos = [];
  rig.cadera.traverse((o) => o.isBone && huesos.push(o));

  if (regiones && (cfg.pelo != null || cfg.remera != null)) {

    if (cfg.pelo != null) {
      u.uColorPelo.value.set(cfg.pelo);
      u.uUsaPelo.value = 1;
    }
    if (cfg.remera != null) {
      u.uColorRemera.value.set(cfg.remera);
      u.uUsaRemera.value = 1;
    }
  }
  const malla = new THREE.SkinnedMesh(g, mat);
  malla.castShadow = true;
  malla.receiveShadow = true;
  malla.frustumCulled = false;
  raiz.add(malla);

  const grosor = cfg.contorno ?? 0.011;
  const contorno = new THREE.SkinnedMesh(gc, materialContornoPiel(grosor));
  contorno.userData.esContorno = true;
  contorno.userData.materialResaltado = materialContornoPiel(grosor * 1.3, 0xffd84a);
  contorno.frustumCulled = false;
  contorno.castShadow = false;
  contorno.raycast = () => {};
  raiz.add(contorno);

  raiz.updateMatrixWorld(true);
  const esqueleto = new THREE.Skeleton(huesos);
  malla.bind(esqueleto);
  contorno.bind(esqueleto);

  // Pose canÃ³nica: brazos colgando (rotaciÃ³n 0, como el placeholder), resto como el modelo
  for (const lado of ['I', 'D']) {

  const manos = dedos.I && dedos.D ? new Manos({ I: rig.brazoI.dedosHuesos, D: rig.brazoD.dedosHuesos }) : null;

  raiz.rotation.copy(rotRaiz);
  raiz.scale.copy(escRaiz);

}

/**
 * Manos con dedos articulados para los modelos con auto-rig.
 *
 * Los GLB traen la mano como un bloque: cuatro dedos rectos y un pulgar, todo pegado a un solo
 * hueso. AcÃ¡ se detectan los dedos en la malla y se arma, por mano, un esqueleto de
 *   pulgar (3 huesos) + Ã­ndice, medio, anular y meÃ±ique (3 huesos cada uno)
 * colgado del hueso "mano". Con eso la mano puede quedar relajada, hacer puÃ±o, seÃ±alar,
 * agarrar una lapicera o mostrar de 1 a 4 dedos (las seÃ±as de Sofi) sin manos "postizas".
 *
 * ConvenciÃ³n (igual que el resto del rig): los huesos de los dedos nacen con la orientaciÃ³n
 * del antebrazo; el eje Z local es el eje de flexiÃ³n (los dedos se cierran hacia la palma).
 */

export const DEDOS = ['pulgar', 'indice', 'medio', 'anular', 'menique'];

const _a = new THREE.Vector3();
const _b = new THREE.Vector3();
const _c = new THREE.Vector3();

// ---------- DetecciÃ³n ----------

/**
 * Detecta los dedos de UNA mano en la geometrÃ­a normalizada (mirando a -Z, pose A).
 * Devuelve null si la mano no tiene la forma esperada (en ese caso se usa el modo viejo).
 *   art  articulaciones de detectarArticulaciones()
 *   lado 'I' | 'D'
 * Resultado: { dedos: [{ nombre, puntos:[p0,p1,p2,p3] }], eje, largo }
 *   p0 = nudillo (base del dedo), p3 = punta; p1 y p2 son las articulaciones intermedias.
 */
export function detectarDedos(pos, art, lado, H, diag = []) {
  const signo = lado === 'I' ? -1 : 1;
  const mu = art['muneca' + lado];
  const pu = art['punta' + lado];
  const hombro = art['hombro' + lado];
  const u = new THREE.Vector3().subVectors(pu, mu);
  const L = u.length();
  if (L < H * 0.05) return diag.push('mano demasiado corta'), null;
  u.normalize();

  // VÃ©rtices de la mano: pasada la muÃ±eca, cerca del eje y del lado correcto
  const xMin = Math.abs(hombro.x) * 0.55;
  const mano = [];
  for (let i = 0; i < pos.count; i++) {
    const x = pos.getX(i);
    if (x * signo < xMin) continue;
    _a.fromBufferAttribute(pos, i).sub(mu);
    const t = _a.dot(u);
    if (t < 0.05 * L || t > 1.08 * L) continue;
    const radial = _b.copy(_a).addScaledVector(u, -t).length();
    if (radial > 0.62 * L) continue;
    mano.push({ i, t: t / L, c: _a.z, p: new THREE.Vector3().fromBufferAttribute(pos, i) });
  }
  if (mano.length < 60) return diag.push(`pocos vÃ©rtices de mano (${mano.length})`), null;

  // Puntas de los cuatro dedos largos: k-means (k = 4) sobre la coordenada transversal (z)
  // de los vÃ©rtices de la franja distal. El pulgar no llega hasta ahÃ­.
  const distal = mano.filter((q) => q.t > 0.74);
  if (distal.length < 20) return diag.push(`pocos vÃ©rtices en la franja de puntas (${distal.length})`), null;
  const cs = distal.map((q) => q.c).sort((x, y) => x - y);
  let centros = [0.125, 0.375, 0.625, 0.875].map((f) => cs[Math.floor(f * (cs.length - 1))]);
  let grupo = new Array(distal.length).fill(0);
  for (let it = 0; it < 25; it++) {
    distal.forEach((q, n) => {
      let mejor = 0;
      for (let k = 1; k < 4; k++) if (Math.abs(q.c - centros[k]) < Math.abs(q.c - centros[mejor])) mejor = k;
      grupo[n] = mejor;
    });
    const suma = [0, 0, 0, 0];
    const cant = [0, 0, 0, 0];
    distal.forEach((q, n) => {
      suma[grupo[n]] += q.c;
      cant[grupo[n]]++;
    });
    for (let k = 0; k < 4; k++) if (cant[k]) centros[k] = suma[k] / cant[k];
  }
  // Ordenados de adelante (-z, junto al pulgar) hacia atrÃ¡s: Ã­ndice, medio, anular, meÃ±ique
  const orden = [0, 1, 2, 3].sort((x, y) => centros[x] - centros[y]);
  const largos = [0.4, 0.43, 0.4, 0.33]; // largo de cada dedo respecto del largo de la mano

  const dedos = [];
  const puntaDe = (miembros) => {
    // punta = promedio de los vÃ©rtices mÃ¡s lejanos del grupo
    const ord = [...miembros].sort((q1, q2) => q2.t - q1.t);
    const n = Math.max(3, Math.min(8, Math.floor(ord.length * 0.15)));
    const p = new THREE.Vector3();
    for (let k = 0; k < n; k++) p.add(ord[k].p);
    return p.multiplyScalar(1 / n);
  };
  for (let f = 0; f < 4; f++) {
    const miembros = distal.filter((_, n) => grupo[n] === orden[f]);
    if (miembros.length < 3) return diag.push(`dedo ${f + 1} sin vÃ©rtices`), null;
    const punta = puntaDe(miembros);
    const largo = Math.max(L * 0.28, Math.min(L * 0.6, punta.distanceTo(mu) * largos[f]));
    const dir = _c.subVectors(punta, mu).normalize();
    const base = punta.clone().addScaledVector(dir, -largo);
    const j1 = base.clone().lerp(punta, 0.4);
    const j2 = base.clone().lerp(punta, 0.72);
    dedos.push({ nombre: DEDOS[f + 1], puntos: [base, j1, j2, punta.clone()] });
  }

  // Pulgar: vÃ©rtices adelante (-z) del eje del Ã­ndice, sin llegar a la franja de las puntas
  const indice = dedos[0];
  const tIndice = _a.subVectors(indice.puntos[3], mu).dot(u) / L;
  const cIndice = indice.puntos[3].z - mu.z;
  const anchoDedo = Math.abs(centros[orden[1]] - centros[orden[0]]) || L * 0.12;
  const pulgarPts = mano.filter((q) => q.t < 0.8 && q.c < cIndice * (q.t / Math.max(0.3, tIndice)) - anchoDedo * 0.6);
  if (pulgarPts.length < 15) return diag.push(`pulgar sin vÃ©rtices (${pulgarPts.length})`), null;
  // La punta del pulgar: el extremo mÃ¡s adelantado (-z) y largo del conjunto
  const ranking = pulgarPts
    .filter((q) => q.t > 0.3)
    .map((q) => ({ q, v: -q.c / L + 0.35 * q.t }))
    .sort((x, y) => y.v - x.v);
  if (ranking.length < 6) return diag.push('pulgar sin punta'), null;
  const punta = new THREE.Vector3();
  for (let k = 0; k < 6; k++) punta.add(ranking[k].q.p);
  punta.multiplyScalar(1 / 6);
  const tp = punta.clone();
  const baseP = mu.clone().lerp(tp, 0.3);
  dedos.unshift({
    nombre: 'pulgar',
    puntos: [baseP, baseP.clone().lerp(tp, 0.42), baseP.clone().lerp(tp, 0.74), tp],
  });

  return { dedos, eje: u, largo: L, mano };
}

/** Crea los huesos de los dedos de una mano. Devuelve { pulgar: [h1,h2,h3], indice: [...], ... }. */
export function crearHuesosDedos(manoHueso, deteccion, lado) {
  const huesos = {};
  manoHueso.updateMatrixWorld(true);
  for (const d of deteccion.dedos) {
    const cadena = [];
    let padre = manoHueso;
    for (let k = 0; k < 3; k++) {
      const b = new THREE.Bone();
      b.name = `${d.nombre}${k + 1}${lado}`;
      padre.add(b);
      padre.updateMatrixWorld(true);
      b.position.copy(padre.worldToLocal(d.puntos[k].clone()));
      b.updateMatrixWorld(true);
      cadena.push(b);
      padre = b;
    }
    huesos[d.nombre] = cadena;
  }
  return huesos;
}

function distSeg(p, a, b, salida) {
  const ab = _b.subVectors(b, a);
  const t = THREE.MathUtils.clamp(_a.subVectors(p, a).dot(ab) / Math.max(1e-9, ab.lengthSq()), 0, 1);
  const q = _c.copy(a).addScaledVector(ab, t);
  if (salida) salida.t = t;
  return p.distanceTo(q);
}

const suave = (x) => {
  const t = THREE.MathUtils.clamp(x, 0, 1);
  return t * t * (3 - 2 * t);
};

/**
 * Sobrescribe los pesos de piel de los vÃ©rtices de los dedos.
 *   deteccion   resultado de detectarDedos()
 *   huesosDedos resultado de crearHuesosDedos()
 *   indiceHueso (hueso) => Ã­ndice en el esqueleto
 *   manoHueso   hueso "mano" (para la uniÃ³n con la palma)
 */
export function pesosDedos(g, deteccion, huesosDedos, indiceHueso, manoHueso, L) {
  const pos = g.attributes.position;
  const idx = g.attributes.skinIndex;
  const pes = g.attributes.skinWeight;
  const p = new THREE.Vector3();
  const aux = { t: 0 };
  const manoIdx = indiceHueso(manoHueso);
  const radio = L * 0.2; // ancho mÃ¡ximo de un dedo (desde su eje)
  let tocados = 0;
  for (const q of deteccion.mano) {
    p.fromBufferAttribute(pos, q.i);
    // dedo mÃ¡s cercano (distancia al segmento nudillo-punta, normalizada por el grosor)
    let mejor = null;
    let mejorD = Infinity;
    for (const d of deteccion.dedos) {
      const grosor = d.nombre === 'pulgar' ? radio * 1.25 : radio * 0.8;
      const dist = distSeg(p, d.puntos[0], d.puntos[3], aux) / grosor;
      if (dist < mejorD) {
        mejorD = dist;
        mejor = { d, s: aux.t };
      }
    }
    if (!mejor || mejorD > 1.15) continue;
    // s: 0 en el nudillo, 1 en la punta. Para vÃ©rtices antes del nudillo no hay proyecciÃ³n
    // negativa (clamp), asÃ­ que se mira tambiÃ©n la posiciÃ³n a lo largo del eje de la mano.
    const d = mejor.d;
    const cadena = huesosDedos[d.nombre];
    const ab = _b.subVectors(d.puntos[3], d.puntos[0]);
    const sReal = _a.subVectors(p, d.puntos[0]).dot(ab) / Math.max(1e-9, ab.lengthSq());
    const w1 = 1 - suave((sReal - 0.4 + 0.1) / 0.2); // falange proximal
    const w3 = suave((sReal - 0.72 + 0.1) / 0.2); // falange distal
    let wProx = w1;
    let wDist = w3;
    let wMed = Math.max(0, 1 - wProx - wDist);
    // uniÃ³n con la palma: antes y alrededor del nudillo se mezcla con la mano
    const wPalma = 1 - suave((sReal + 0.1) / 0.28);
    const resto = 1 - wPalma;
    wProx *= resto;
    wMed *= resto;
    wDist *= resto;
    const lista = [
      [manoIdx, wPalma],
      [indiceHueso(cadena[0]), wProx],
      [indiceHueso(cadena[1]), wMed],
      [indiceHueso(cadena[2]), wDist],
    ];
    let total = 0;
    for (const [, w] of lista) total += w;
    if (total < 1e-6) continue;
    for (let k = 0; k < 4; k++) {
      idx.setComponent(q.i, k, lista[k][0]);
      pes.setComponent(q.i, k, lista[k][1] / total);
    }
    tocados++;
  }
  return tocados;
}

/** Pose de seÃ±as: se estiran los primeros n dedos (1 = Ã­ndice â€¦ 4 = meÃ±ique), el resto en puÃ±o. */
export function poseSenia(n) {
  const base = {};
  const abiertos = ['indice', 'medio', 'anular', 'menique'];
  const separacion = { indice: -0.1, medio: -0.01, anular: 0.09, menique: 0.21 }; // bien abiertos para contar
  for (const [nombre, def] of Object.entries(POSES_MANO.puno)) base[nombre] = def;
  abiertos.forEach((nombre, i) => {
    if (i < n) base[nombre] = { f: [0.02, 0.02, 0.02], abrir: separacion[nombre] };
  });
  base.pulgar = { f: [0.3, 0.45, 0.4], abrir: 0.0, cruzar: 0.85 };
  return base;
}

/**
 * Mueve los huesos de los dedos de un personaje con suavizado. Se crea una vez por personaje con modelo.
 *   huesos: { I: { pulgar: [..3], indice: [..3], ... }, D: {...} }
 */
export class Manos {
  constructor(huesos) {
    this.huesos = huesos;
    this.estado = { I: this.copiar(POSES_MANO.reposo), D: this.copiar(POSES_MANO.reposo) };
    this.objetivo = { I: POSES_MANO.reposo, D: POSES_MANO.reposo };
    this.forzada = { I: null, D: null };
    this.velocidad = 14;
    this.aplicar(0, true);
  }

  copiar(pose) {
    const r = {};
    for (const [nombre, d] of Object.entries(pose)) r[nombre] = { f: [...d.f], abrir: d.abrir ?? 0, cruzar: d.cruzar ?? 0 };
    return r;
  }

  /** Pose de mano asociada a la pose del brazo (se ignora mientras haya una pose forzada). */
  pose(lado, nombre) {
    this.objetivo[lado] = POSES_MANO[nombre] ?? POSES_MANO.reposo;
  }

  /** Fuerza una pose (por ejemplo las seÃ±as); con null se vuelve a la pose del brazo. */
  forzar(lado, pose) {
    this.forzada[lado] = pose;
  }

  update(dt) {
    this.aplicar(dt, false);
  }

  aplicar(dt, inmediato) {
    const k = inmediato ? 1 : 1 - Math.exp(-this.velocidad * dt);
    for (const lado of ['I', 'D']) {
      const meta = this.forzada[lado] ?? this.objetivo[lado];
      const actual = this.estado[lado];
      // ConvenciÃ³n de signo: derecha cierra con z negativo, izquierda con z positivo
      const signo = lado === 'D' ? -1 : 1;
      for (const nombre of DEDOS) {
        const cadena = this.huesos[lado][nombre];
        if (!cadena) continue;
        const m = meta[nombre] ?? POSES_MANO.reposo[nombre];
        const a = actual[nombre];
        for (let i = 0; i < 3; i++) a.f[i] += (m.f[i] - a.f[i]) * k;
        a.abrir += ((m.abrir ?? 0) - a.abrir) * k;
        a.cruzar += ((m.cruzar ?? 0) - a.cruzar) * k;
        for (let i = 0; i < 3; i++) {
          if (nombre === 'pulgar') {
            // El pulgar se cierra cruzando la palma: flexiÃ³n (z) + giro hacia los dedos (x)
            cadena[i].rotation.set(i === 0 ? -a.cruzar : -a.cruzar * 0.25, 0, signo * a.f[i]);
          } else {
            cadena[i].rotation.set(0, 0, signo * a.f[i]);
          }
        }
        // apertura lateral en la primera falange (rotaciÃ³n alrededor del eje de la palma)
        if (nombre !== 'pulgar') cadena[0].rotation.x = a.abrir;
        else cadena[0].rotation.y = signo * a.abrir;
      }
    }
  }
}
